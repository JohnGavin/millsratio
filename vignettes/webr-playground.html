## Quick Start Examples

### Example 1: Basic Mills Ratio Calculation

Try calculating Mills ratios for different distributions. Click "Run Code" to execute:

```{webr-r}
# Calculate Mills ratio for normal distribution
x <- seq(1, 5, by = 0.5)
m_normal <- mills_ratio_normal(x)
print(round(m_normal, 3))

# Calculate for t-distribution with 30 df
m_t30 <- mills_ratio_t(x, df = 30)
print(round(m_t30, 3))

# Compare the ratios
ratio <- m_t30 / m_normal
cat("\nt(30)/Normal ratio:\n")
print(round(ratio, 3))
```

### Example 2: The t(30) Paradox

Explore why t(30) appears normal but has fatter tails:

```{webr-r}
# Analyze the t(30) paradox
paradox_data <- analyze_t30_paradox()

# Show the divergence
cat("t(30) vs Normal Mills Ratio Divergence:\n")
cat("=====================================\n")
for(i in 1:nrow(paradox_data)) {
  cat(sprintf("x = %.1f: Normal = %.3f, t(30) = %.3f, Ratio = %.3f\n",
              paradox_data$x[i],
              paradox_data$normal[i],
              paradox_data$t30[i],
              paradox_data$ratio[i]))
}

# Key insight
cat("\nKey insight: At x=4, t(30) Mills ratio is 54% higher than normal!\n")
```

### Example 3: Tidyverse Analysis

Use dplyr and tidyr for Mills ratio analysis:

```{webr-r}
library(dplyr)
library(tidyr)

# Create comparison data frame
comparison <- tibble(x = seq(0.5, 3, by = 0.5)) %>%
  mutate(
    Normal = mills_ratio_normal(x),
    `t(30)` = mills_ratio_t(x, df = 30),
    `t(3)` = mills_ratio_t(x, df = 3),
    Exponential = mills_ratio_exp(x)
  )

# Display the comparison
print(comparison)

# Compute summary statistics
comparison %>%
  pivot_longer(-x, names_to = "distribution", values_to = "mills_ratio") %>%
  group_by(distribution) %>%
  summarise(
    mean = mean(mills_ratio),
    sd = sd(mills_ratio),
    min = min(mills_ratio),
    max = max(mills_ratio)
  ) %>%
  print()
```

### Example 4: Hazard Function Analysis

Explore the relationship between Mills ratio and hazard function:

```{webr-r}
# Calculate hazard from Mills ratio
x <- c(1, 2, 3, 4, 5)
mills <- mills_ratio_normal(x)
hazard <- hazard_from_mills(mills)

# Create comparison table
results <- data.frame(
  x = x,
  mills_ratio = round(mills, 4),
  hazard = round(hazard, 4),
  product = round(mills * hazard, 4)
)

print(results)
cat("\nNote: Mills ratio × Hazard = 1 (they are reciprocals)\n")

# Compare hazard functions across distributions
x_seq <- seq(0.5, 5, by = 0.5)
hazard_comparison <- data.frame(
  x = x_seq,
  normal = hazard_function(x_seq, "normal"),
  t30 = hazard_function(x_seq, "t", df = 30),
  exponential = hazard_function(x_seq, "exponential")
)

cat("\nHazard Function Comparison:\n")
print(head(hazard_comparison, 10))
```

### Example 6: Verify Mills Ratio Properties

Test the theoretical properties of Mills ratios:

```{webr-r}
# Test 1: Normal has decreasing Mills ratio (thin tails)
x_test <- seq(1, 5, by = 0.5)
m_normal <- mills_ratio_normal(x_test)
normal_decreasing <- all(diff(m_normal) < 0)
cat("Normal Mills ratio is decreasing:", normal_decreasing, "\n")

# Test 2: t(3) has increasing Mills ratio (fat tails)
m_t3 <- mills_ratio_t(x_test, df = 3)
t3_increasing <- all(diff(m_t3) > 0)
cat("t(3) Mills ratio is increasing:", t3_increasing, "\n")

# Test 3: Exponential has constant Mills ratio
m_exp <- mills_ratio_exp(x_test)
exp_constant <- all(abs(diff(m_exp)) < 1e-10)
cat("Exponential Mills ratio is constant:", exp_constant, "\n")

# Test 4: Hazard function properties
h_normal <- hazard_function(x_test, "normal")
normal_hazard_increasing <- all(diff(h_normal) > 0)
cat("\nNormal hazard is increasing (IFR):", normal_hazard_increasing, "\n")

# Display the values
cat("\nMills Ratio Values:\n")
results <- data.frame(
  x = x_test,
  Normal = round(m_normal, 3),
  `t(3)` = round(m_t3, 3),
  Exponential = round(m_exp, 3)
)
print(results)
```

## Advanced Example: Monte Carlo Verification

Verify Mills ratio calculations using simulation:

```{webr-r}
# Monte Carlo verification of Mills ratio
set.seed(123)
n_sim <- 10000
x_val <- 2

# Simulate from standard normal
samples <- rnorm(n_sim)

# Empirical Mills ratio
survivors <- samples[samples > x_val]
empirical_mills <- length(survivors) / n_sim / dnorm(x_val)

# Theoretical Mills ratio
theoretical_mills <- mills_ratio_normal(x_val)

# Compare
cat("Monte Carlo Verification (n =", n_sim, "simulations)\n")
cat("========================================\n")
cat("x value:", x_val, "\n")
cat("Empirical Mills ratio:", round(empirical_mills, 4), "\n")
cat("Theoretical Mills ratio:", round(theoretical_mills, 4), "\n")
cat("Relative error:", round(abs(empirical_mills - theoretical_mills) / theoretical_mills * 100, 2), "%\n")

# Try with t-distribution
t_samples <- rt(n_sim, df = 30)
t_survivors <- t_samples[t_samples > x_val]
empirical_t_mills <- length(t_survivors) / n_sim / dt(x_val, df = 30)
theoretical_t_mills <- mills_ratio_t(x_val, df = 30)

cat("\nt(30) Distribution:\n")
cat("Empirical Mills ratio:", round(empirical_t_mills, 4), "\n")
cat("Theoretical Mills ratio:", round(theoretical_t_mills, 4), "\n")
cat("Relative error:", round(abs(empirical_t_mills - theoretical_t_mills) / theoretical_t_mills * 100, 2), "%\n")
```

## Common Tasks

### Compare Any Two Distributions

```{webr-r}
# Customize this example with your own parameters
x_range <- seq(0.5, 4, by = 0.5)

# Choose your distributions
dist1 <- mills_ratio_normal(x_range)
dist2 <- mills_ratio_t(x_range, df = 10)  # Change df as needed

# Create comparison
comparison <- data.frame(
  x = x_range,
  Distribution1 = dist1,
  Distribution2 = dist2,
  Ratio = dist2 / dist1,
  Percent_Diff = (dist2 / dist1 - 1) * 100
)

print(comparison)

# Find crossover point (if any)
if(any(diff(sign(dist1 - dist2)) != 0)) {
  cat("\nDistributions cross over in the given range!\n")
} else {
  cat("\nNo crossover in the given range.\n")
}
```

### Explore Different Parameters

```{webr-r}
# Explore how degrees of freedom affect t-distribution Mills ratio
x <- 3  # Fixed point
df_values <- c(1, 2, 3, 5, 10, 20, 30, 50, 100, 1000)

results <- data.frame(
  df = df_values,
  mills_ratio = sapply(df_values, function(d) mills_ratio_t(x, df = d)),
  normal_limit = mills_ratio_normal(x)
)

cat("t-distribution Mills ratio at x =", x, "for different df:\n")
cat("=====================================================\n")
print(results)

cat("\nObservation: As df increases, t-distribution approaches normal.\n")
```
